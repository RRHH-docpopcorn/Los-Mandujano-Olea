-- ============================================================ -- Los Mandujano Olea — Finanzas TC -- Ejecutar UNA VEZ en Supabase: Project > SQL Editor > New query -- (mismo proyecto que ya usan para "household_board") -- ============================================================ -- Tabla 1: cada movimiento de la tarjeta de crédito, ya clasificado create table if not exists tc_transacciones ( id text primary key, -- ID único del movimiento (viene del banco) mes text, -- ENERO, FEBRERO, ... fecha date, ciudad text, cuenta text, -- glosa del banco (comercio + ciudad tal cual) total numeric, -- monto total de la compra (si es en cuotas) tt_a_pagar numeric, monto numeric, -- monto de ESTA cuota/movimiento (lo que se usa para sumar el mes) categoria text, -- categoría original (Delivery, Supermercado, etc.) comercio text, -- comercio normalizado (Jumbo, Uber Eats, etc.) grupo text, -- CASA / EDUCACIÓN EMI / AUTO / RECREACIÓN / No Presupuestado item text, -- ítem específico del presupuesto (Supermercado, Bencina, etc.) created_at timestamptz default now() );
